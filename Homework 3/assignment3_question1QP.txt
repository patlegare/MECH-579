import numpy as np
import matplotlib.pyplot as plt

#rosenbrock function
def f(x):
    return (1 - x[0])**2 + 100*(x[1] - x[0]**2)**2

def grad_f(x):
    dfdx = -2*(1 - x[0]) - 400*x[0]*(x[1] - x[0]**2)
    dfdy = 200*(x[1] - x[0]**2)
    return np.array([dfdx, dfdy])

def hess_f(x):
    d2fxx = 2 - 400*x[1] + 1200*x[0]**2
    d2fxy = -400*x[0]
    d2fyy = 200
    return np.array([[d2fxx, d2fxy],
                     [d2fxy, d2fyy]])

#constraints
def c1(x): return 1 - x[0] - x[1]
def grad_c1(x): return np.array([-1, -1])
def hess_c1(x): return np.zeros((2,2))

def c2(x): return 1 - x[0]**2 - x[1]**2
def grad_c2(x): return np.array([-2*x[0], -2*x[1]])
def hess_c2(x): return np.array([[-2, 0], [0, -2]])

#penalty function
def phi(x, mu, c_fun):
    return f(x) + 0.5*mu*(c_fun(x)**2)

def grad_phi(x, mu, c_fun, grad_c_fun):
    return grad_f(x) + mu*c_fun(x)*grad_c_fun(x)

def hess_phi(x, mu, c_fun, grad_c_fun, hess_c_fun):
    return hess_f(x) + mu*np.outer(grad_c_fun(x), grad_c_fun(x)) + mu*c_fun(x)*hess_c_fun(x)

#Backtracking line search
def backtracking(func, x, p, grad, mu, c_fun, alpha=1.0, rho=0.5, c=1e-4):
    fx = func(x, mu, c_fun)
    while func(x + alpha*p, mu, c_fun) > fx + c*alpha*np.dot(grad, p):
        alpha *= rho
    return alpha

#Quadratic penalty method
def quadratic_penalty(x0, c_fun, grad_c_fun, hess_c_fun, mu0=1.0, beta=10, tol=1e-6, max_iter=50):
    x = x0.copy()
    mu = mu0
    grad_norms = []
    trajectory = [x.copy()]

    for k in range(max_iter):
        g = grad_phi(x, mu, c_fun, grad_c_fun)
        H = hess_phi(x, mu, c_fun, grad_c_fun, hess_c_fun)
        
        # Newton step
        p = -np.linalg.solve(H, g)
        alpha = backtracking(phi, x, p, g, mu, c_fun)
        x_new = x + alpha*p

        grad_norms.append(np.linalg.norm(g))
        trajectory.append(x_new.copy())

        # Stopping criteria
        if np.linalg.norm(p) < tol:
            break

        # Increase penalty parameter
        if abs(c_fun(x_new)) > 1e-3:
            mu *= beta

        x = x_new

    return np.array(trajectory), grad_norms, x

#Constraints
x0 = np.array([-1.2, 1.0])
traj1, grad_norms1, x_opt1 = quadratic_penalty(x0, c1, grad_c1, hess_c1)
traj2, grad_norms2, x_opt2 = quadratic_penalty(x0, c2, grad_c2, hess_c2)

# Gradient convergence plot
plt.figure(figsize=(7,5))
plt.semilogy(grad_norms1, label=r'Penalty on $(1-x-y)^2$')
plt.semilogy(grad_norms2, label=r'Penalty on $(1-x^2-y^2)^2$')
plt.xlabel("Iteration", fontsize=12)
plt.ylabel(r"$\log(\|\nabla \Phi(x)\|)$", fontsize=12)
plt.title("Convergence of the Gradient Norm (Quadratic Penalty)")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# contour plot
x_vals = np.linspace(-2, 2, 400)
y_vals = np.linspace(-1, 3, 400)
X, Y = np.meshgrid(x_vals, y_vals)
Z = (1 - X)**2 + 100*(Y - X**2)**2

plt.figure(figsize=(7,6))
plt.contour(X, Y, Z, levels=np.logspace(-1,3,20), cmap='viridis')
plt.plot(traj1[:,0], traj1[:,1], 'r-o', label=r'Path ($1-x-y=0$ penalty)')
plt.plot(traj2[:,0], traj2[:,1], 'b-o', label=r'Path ($1-x^2-y^2=0$ penalty)')
plt.xlabel('x')
plt.ylabel('y')
plt.title('Contour Plot with Quadratic Penalty Optimization Paths')
plt.legend()
plt.tight_layout()
plt.show()
